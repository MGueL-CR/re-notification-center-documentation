blinkIcon: function( blinkTimes, interval, opacity ) {
  
    this.manageAutohide();
    if(blinkTimes > 0) {
      this.notificationIcon.ease({
        duration:   interval,
        opacity:    opacity,
        onComplete: ()=> this.blinkIcon(--blinkTimes,interval,(opacity==255)?100:255)
      });
    }

  }
